func arrays_have_same_content(array1, array2):
	if array1.size() != array2.size(): return false
	for item in array1:
		if !array2.has(item):
			 return false
		if array1.count(item) != array2.count(item):
			 return false
	return true

remote func changeRandomHandCard(number, hp, atk, def, bobux):
	for i in range(0,number):
		var hand = _hand_store.cards()
		if hand.size() < 1:
			return
		hand.shuffle()
		if hand[0].data().has_value("hp"):
			hand[0].data().set_value("hp", hand[0].data().get_value("hp")+hp)
		if hand[0].data().has_value("attack"):
			hand[0].data().set_value("attack", hand[0].data().get_value("attack")+atk)
		hand[0].data().set_value("mana", hand[0].data().get_value("mana")+bobux)
		hand[0].emit_signal("need")
		hand[0].card._update_data(hand[0].data())

	var hand = _hand_store.cards()

	hand.shuffle()

func intersect(array1: Array, array2: Array):
	var intersection = []
	for item in array1:
		if array2.has(item):

func moveCard(base, destination):

		set_carte_variable(base, null)

		rpc("placeCardRemote", base, 0, card.data().get_value("attack"), card.data().get_value("mana"), card.data().get_text("name"), card.data().get_text("desc"), 0)
		reload()
	else:
		var carteDestination = get_store(destination)
		var carteBase = get_store(base)
		
		set_pile_store(base, carteDestination)
		get_pile(base).set_store(carteDestination)
		set_carte_variable(base, get_store(base).cards()[0])
		
		set_pile_store(destination, carteBase)
		get_pile(destination).set_store(carteBase)
		set_carte_variable(destination, get_store(destination).cards()[0])
		
		
		carteBase = get_store(base).cards()[0]

	gameId += $EnnemieName.text

func ouille(row, damage, type):
	var data = get_data(row)
	var nom = data.get_text("name")
	var hp = data.get_value("hp")
	if damage > 0:
		if nom == "41 - Zig" and hp > damage:
			giveCard("41 - Zig", 1)
			effectCard(row, 0, 1000)
		if nom == "72 - Arthuro":
			giveCardTypeMax("poisson", 1, 6)
		if hp == 1 and ennemie_has_card("63 - Herokin"):
			effectCard(row, 0, 1000)

func take_damage(row, damage, type = "physique"):
	if row > 5 or row < 1:
		return
	
	var data = get_data(row)
	
	if has_card("77 - Papa Cerdito") and data.get_category("creature_type") == "pepapig":
		take_damage(returnCards("77 - Papa Cerdito")[0], damage, type)
		return
	
	if data:
		if type == "bouliste":
			hurtOrbe(damage)
		if data.has_value("def"):
			if type != "magic":
				var def = data.get_value("def")
				if def:
					if def > damage:
						pass
					else:
						if type == "karate":
							ouille(row, damage - def + 2, type)
							effectCard(row, 0, damage - def + 2)
						else:
							ouille(row, damage - def, type)
							effectCard(row, 0, damage - def)
				else:
					ouille(row, damage, type)
					effectCard(row, 0, damage)
			else:
				effectCard(row, 0, damage)
		else:
			ouille(row, damage, type)
			effectCard(row, 0, damage)
	else:
		if type != "noharam":
			if type == "chob":
				hurtOrbe(2)
			hurtOrbe(damage)
	
	if data:
		if data.get_value("hp") == 1 and data.get_text("name") == "69 - Maliki":
			effectCard(row, -2, -2)
		

remote func deal_damage(row, damage, type = "physique"):
	take_damage(row, damage, type)

func placeCard(row, name):
	if get_store(row).cards().size() == 0:
		var db = CardEngine.db().get_database("main")
		var q = Query.new()
		var cards = q.contains(["name:"+name]).execute(db)

		var store = CardPile.new()
		
		store.populate(db, cards)
		store.keep(1)
		get_pile(row).set_store(store)
		set_pile_store(row, get_pile(row).store())
		_hand_store.play_card(get_store(row).cards()[0].ref(), get_pile(row))
		set_carte_variable(row, get_store(row).cards()[0])
		return get_store(row).cards()[0]
	return null

func effectCard(index, atk, hp, def = 0):
	if get_store(index).cards().size() > 0:
		if get_data(index).get_text("name") == "31 - Requin Cerbère" and atk < 0:
			return
		get_data(index).set_value("hp", get_data(index).get_value("hp")-hp)
		get_data(index).set_value("attack", get_data(index).get_value("attack")-atk)
		particle(index, atk, hp)
		if atk <= 0 and hp <= 0:
			if get_data(index).get_text("name") != "115 - Ninja Kong":
				for i in returnCards("115 - Ninja Kong"):
					get_data(i).set_value("hp", get_data(i).get_value("hp")+1)
					get_data(i).set_value("attack", get_data(i).get_value("attack")+1)
					particle(i, atk, hp)
		if atk <= 0 or hp <= 0 or def <= 0:
			if get_data(index).get_text("name") == "81 - Peppa Pig":
				var freeSpace = []
				if get_store(1).cards().size() != 0 and index != 1:
					if get_data(1).get_text("name") != "81 - Peppa Pig":
						freeSpace.append(1)
				if get_store(2).cards().size() != 0 and index != 2:
					if get_data(2).get_text("name") != "81 - Peppa Pig":
						freeSpace.append(2)
				if get_store(3).cards().size() != 0 and index != 3:
					if get_data(3).get_text("name") != "81 - Peppa Pig":
						freeSpace.append(3)
				if get_store(4).cards().size() != 0 and index != 4:
					if get_data(4).get_text("name") != "81 - Peppa Pig":
						freeSpace.append(4)
				if get_store(5).cards().size() != 0 and index != 5:
					if get_data(5).get_text("name") != "81 - Peppa Pig":
						freeSpace.append(5)
				
				if freeSpace.size() > 0:
					freeSpace.shuffle()
					if hp < 0 or atk < 0:
						effectCard(freeSpace[0], atk, hp)
					if def < 0:
						cardDef(freeSpace[0], def)
		
	reload()
	check_death()
